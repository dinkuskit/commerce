    ManagedSkuRegistrationError,
    confirmExistingManagedSku,

  const cases = [

      input: {
        mode: "managed",
        status: "active",
        inventorySkuId: " inventory-sku-1 ",
      },

      input: { mode: "managed", status: "active" },
    },
    {
      name: "malformed-review",
      input: {
        mode: "managed",
        status: "needs-review",
        candidate: {
          inventorySkuId: "",
          sku: "GRILL-1",
          displayName: "Demo Grill",
        },
      },
    },
  ];

  for (const value of cases.slice(0, 2)) {
    console.log(JSON.stringify({
      case: value.name,
      result: normalizeStoredStockManagement(value.input),
    }));
  }

  try {
    confirmExistingManagedSku(cases[2].input);
  } catch (error) {
    console.log(JSON.stringify({
      case: "malformed-review-confirmation",
      error: {
        name: error.name,
        code: error.code,
        message: error.message,
        isManagedSkuRegistrationError:
          error instanceof ManagedSkuRegistrationError,
      },
    }));
  }
'
